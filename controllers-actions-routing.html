    import { Controller, Get, Param, Post } from '@nestjs/common';

@Controller('premier')
export class PremierController {
   // GET /first
  @Get()
  index(): string {
    return 'Action [index] du contrôleur [premier]';
  }

   // GET /first/action01
  @Get('action01')
  action01(): string {
    return 'Action [action01] du contrôleur [premier], requête GET';
  }

   // POST /first/action01: same path, different method HTTP
  @Post('action01')
  action01Post(): string {
    return 'Action [action01Post] du contrôleur [premier], requête POST';
  }

   // GET /first/products/17: [:id] is a route parameter
  @Get('produits/:id')
  produit(@Param('id') id: string): string {
    return `Produit n° ${id}`;
  }

   // GET /first/products/17/reviews/3: two route parameters
  @Get('produits/:id/avis/:numero')
  avis(@Param() params: { id: string; numero: string }): string {
    return `Avis n° ${params.numero} du produit n° ${params.id}`;
  }

   // GET /first/files/... : [*] accepts any path ending
  @Get('fichiers/*')
  fichier(@Param('0') chemin: string): string {
    return `Fichier demandé : ${chemin}`;
  }
}

    Let’s break down this code:

    • line 3: [@Controller('premier')] sets the prefix common to all routes in the controller: its actions handle URL routes that begin with [/premier];
    • lines 6–7: [@Get()] with no arguments: the action [index] handles [GET /premier];
    • lines 12–13: the action [action01] processes [GET /premier/action01]. URL is the concatenation of the controller prefix and the action path;
    • lines 18–19: the same path, but for the POST method: [@Post] replaces [@Get]. There are also [@Put], [@Delete], [@Patch], [@All]... The method name [TypeScript] ([action01Post]) is arbitrary: only the decorators matter for routing;
    • Lines 24–25: [:id] is a route parameter: this segment of URL can take any value. The [@Param('id')] decorator injects its value into the action’s [id] parameter. This value is always a string: for [/premier/produits/17], [id] is ['17'], not the number 17;
    • lines 30–31: two route parameters. [@Param()] without an argument injects an object that contains them all: [{ id: '17', numero: '3' }];
    • lines 36–37: the character [*] accepts any ending of URL, including [/]. The corresponding part is the parameter named [0].

    import { Controller, Get, Header, HttpCode, NotFoundException } from '@nestjs/common';

@Controller('reponses')
export class ReponsesController {
   // a string: sent as-is, with the type [text/html]
  @Get('html')
  html(): string {
    return '<h1>Action [html]</h1>';
  }

   // the same string, but declared as plain text
  @Get('texte')
  @Header('Content-Type', 'text/plain; charset=utf-8')
  texte(): string {
    return '<h1>Action [texte]</h1>';
  }

   // an object: serialized as JSON, with the type [application/json]
  @Get('json')
  json(): object {
    return { nom: 'Martin', prenom: 'Jean', age: 42 };
  }

   // an array: also serialized as JSON
  @Get('tableau')
  tableau(): string[] {
    return ['Mme', 'Mlle', 'Mr'];
  }

   // a selected status code: 201 (Created) instead of 200 (OK)
  @Get('cree')
  @HttpCode(201)
  cree(): object {
    return { message: 'ressource créée' };
  }

   // no response: code 204 (No Content) means “nothing to display”
  @Get('vide')
  @HttpCode(204)
  vide(): void {}

   // an asynchronous method: NestJS waits for the promise to resolve
  @Get('promesse')
  async promesse(): Promise<object> {
    const debut = Date.now();
    await new Promise((fini) => setTimeout(fini, 500));
    return { attente: `${Date.now() - debut} ms` };
  }

   // an exception: NestJS converts it into an error response
  @Get('erreur')
  erreur(): string {
    throw new NotFoundException('Cette ressource n\'existe pas');
  }
}

    Let’s break down this code:

    • lines 6–9: a string is sent as-is, with the header [Content-Type: text/html; charset=utf-8];
    • lines 12–16: the [@Header] decorator sets a response header. Declared as plain text ([text/plain]), the same string is no longer interpreted by the browser;
    • lines 19–22: an object is serialized as JSON, with the header [Content-Type: application/json];
    • lines 25–28: an array is also serialized as JSON;
    • lines 31–35: [@HttpCode(201)] changes the response status code (default 200 for a GET);
    • lines 38–40: the code 204 (No Content) means “no content”: the action returns nothing;
    • lines 43–48: an action can be asynchronous ([async]): in that case, it returns a promise ([Promise]), and [NestJS] waits for it to resolve before responding. This is the case for all actions that query a database;
    • Lines 51–54: An action can trigger an exception. The exceptions for [NestJS] ([NotFoundException], [BadRequestException], [ForbiddenException]...) each carry a status code HTTP (here 404), which [NestJS] uses for its response.

    import { Controller, Get, Param, Redirect } from '@nestjs/common';

@Controller()
export class RedirectionsController {
   // fixed redirection: the URL and code are known in advance
  @Get('ancienne')
  @Redirect('/nouvelle', 301)
  ancienne(): void {}

  @Get('nouvelle')
  nouvelle(): string {
    return 'Vous êtes sur la page [nouvelle]';
  }

   // calculated redirection: the rendered object replaces the address of @Redirect()
  @Get('aller/:destination')
  @Redirect()
  aller(@Param('destination') destination: string): { url: string } {
    return { url: `/page/${destination}` };
  }

  @Get('page/:nom')
  page(@Param('nom') nom: string): string {
    return `Vous êtes sur la page [${nom}]`;
  }

   // redirect to another site
  @Get('documentation')
  @Redirect('https://docs.nestjs.com')
  documentation(): void {}
}

    Let’s break down this code:

    • lines 6–8: [@Redirect('/nouvelle', 301)]: a permanent redirect. The 301 code (Moved Permanently) indicates that the page has permanently changed its address; the default code is 302 (Found), a temporary redirect;
    • lines 16–20: a calculated redirect: [@Redirect()] without an argument, and the action returns an object [{ url: ... }] that sets the destination address;
    • lines 28–30: You can redirect to another site.

    import { Controller, Get, Req, Res } from '@nestjs/common';
import { Request, Response } from 'express';

@Controller('express')
export class ExpressController {
   // The Express request: method, URL, headers...
  @Get('requete')
  requete(@Req() req: Request): object {
    return {
      methode: req.method,
      url: req.url,
      chemin: req.path,
      navigateur: req.get('user-agent'),
    };
  }

   // the Express response: the action itself sends the response
  @Get('reponse')
  reponse(@Res() res: Response): void {
    res.status(200).type('text/plain').send('Réponse envoyée par [res.send]');
  }

   // redirection with the Express response
  @Get('redirection')
  redirection(@Res() res: Response): void {
    res.redirect('/express/reponse');
  }

   // passthrough: the response is used (here for a header), but NestJS
   // always sends the value returned by the action itself
  @Get('passthrough')
  passthrough(@Res({ passthrough: true }) res: Response): string {
    res.setHeader('X-Exemple', '05-express');
    return 'Réponse envoyée par NestJS, avec un entête ajouté';
  }

   // ERREUR—do not do this: @Res() without passthrough, and nothing is sent
  @Get('bloquee')
  bloquee(@Res() res: Response): string {
    return 'Cette chaîne ne sera jamais envoyée : le navigateur attend...';
  }
}

    Let’s break down this code:

    • line 2: the types [Request] and [Response] from [Express];
    • Lines 7–15: [@Req()] injects the request [Express]: method HTTP, URL, path, headers ([req.get(...)])... ;

    import { Injectable } from '@nestjs/common';

export interface Produit {
  id: number;
  nom: string;
  prix: number;
}

@Injectable()
export class ProduitsService {
  private readonly produits: Produit[] = [
    { id: 1, nom: 'stylo', prix: 1.5 },
    { id: 2, nom: 'cahier', prix: 3.2 },
    { id: 3, nom: 'gomme', prix: 0.8 },
  ];

  findAll(): Produit[] {
    return this.produits;
  }

  findById(id: number): Produit | undefined {
    return this.produits.find((p) => p.id === id);
  }
}

    • lines 11–15: the data, here in memory;
    • lines 17–23: the service’s two methods.

    import { Controller, Get, NotFoundException, Param } from '@nestjs/common';
import { Produit, ProduitsService } from './produits.service';

@Controller('produits')
export class ProduitsController {
   // dependency injection: NestJS provides the instance of ProduitsService
  constructor(private readonly produitsService: ProduitsService) {}

  @Get()
  liste(): Produit[] {
    return this.produitsService.findAll();
  }

  @Get(':id')
  unProduit(@Param('id') id: string): Produit {
    const produit = this.produitsService.findById(Number(id));
    if (!produit) {
      throw new NotFoundException(`Produit n° ${id} introuvable`);
    }
    return produit;
  }
}
    • line 7: dependency injection. The controller declares in its constructor that it needs a [ProduitsService]; [NestJS] provides it with the single instance of the service. The controller never calls [new ProduitsService()]. The syntax [private readonly produitsService: ProduitsService] of [TypeScript] declares both the parameter and the field [this.produitsService];
    • Lines 9–12: The action delegates to the service;
    • lines 14–21: [Number(id)] converts the route parameter (a string) into a number. A non-existent product triggers a 404 exception.
