    5. Internationalization

    An internationalized application displays its text in the user’s language. The case study is in French and English. The principle is simple: the views no longer contain text, but keys ([MENU.ACCUEIL]), and a dictionary for each language associates a text with each key. The selected language is stored in a cookie.

    import { readFileSync } from 'fs';
import { join } from 'path';

export type Langue = 'fr' | 'en';
export const LANGUES: Langue[] = ['fr', 'en'];
export const LANGUE_PAR_DEFAUT: Langue = 'fr';
// the "locale" used by Intl for dates and numbers
export const LOCALES: Record<Langue, string> = { fr: 'fr-FR', en: 'en-US' };

// A dictionary is a nested JSON object: { "MENU": { "ACCUEIL": "Home" } }
type Dictionnaire = { [cle: string]: string | Dictionnaire };

// Dictionaries are read UNE times at startup
const DOSSIER = join(__dirname, '..', 'i18n');
const dictionnaires: Record<Langue, Dictionnaire> = {
  fr: JSON.parse(readFileSync(join(DOSSIER, 'fr.json'), 'utf-8')),
  en: JSON.parse(readFileSync(join(DOSSIER, 'en.json'), 'utf-8')),
};

export function estUneLangue(valeur: unknown): valeur is Langue {
  return typeof valeur === 'string' && (LANGUES as string[]).includes(valeur);
}

// translate('fr', 'ACCUEIL.BONJOUR', { name: 'Marie' }) -> "Hello Marie!"
export function traduire(langue: Langue, cle: string, params: Record<string, unknown> = {}): string {
  const texte = chercher(dictionnaires[langue], cle) ?? cle;
  return texte.replace(/\{(\w+)\}/g, (tout, nom: string) => (nom in params ? String(params[nom]) : tout));
}

// 'ACCUEIL.BONJOUR': we go down to ACCUEIL, then to BONJOUR
function chercher(dictionnaire: Dictionnaire, cle: string): string | null {
  let noeud: string | Dictionnaire | undefined = dictionnaire;
  for (const morceau of cle.split('.')) {
    if (typeof noeud !== 'object') {
      return null;
    }
    noeud = noeud[morceau];
  }
  return typeof noeud === 'string' ? noeud : null;
}

    Let’s comment on this code:

    • lines 4–6: the supported languages;
    • line 8: the “locale” for each language, used by [Intl] to format dates and numbers ([fr-FR]: 09/23/2026, €12.50; [en-US]: 9/23/2026, €12.50);
    • lines 14–18: the dictionaries are read once, when the module loads, and then kept in memory;
    • lines 20–22: a check for a type such as [TypeScript] ([valeur is Langue]): after a positive test, the compiler knows that the value is a [Langue];
    • lines 25–28: the translation. The key is looked up in the dictionary; if it is not found, the key itself is displayed: any omission is immediately visible on the screen. The regular expression [/\{(\w+)\}/g] finds each parameter [{nom}] and replaces it with its value;
    • lines 31–40: The key [ACCUEIL.BONJOUR] is split into [ACCUEIL] and then [BONJOUR], and the process continues down into the object JSON.

    import { HelperOptions } from 'handlebars';
import { Langue, LANGUE_PAR_DEFAUT, LOCALES, estUneLangue, traduire } from './i18n';

interface Hbs {
  registerHelper(nom: string, fonction: (...args: any[]) => unknown): void;
}

// the page's language: res.locals.lang, merged into the root template
function langueDe(options: HelperOptions): Langue {
  const lang = options.data?.root?.lang;
  return estUneLangue(lang) ? lang : LANGUE_PAR_DEFAUT;
}

export function enregistrerHelpers(hbs: Hbs): void {
   // {{t 'CLE'}} or {{t 'CLE' name=...}}: translation, with named parameters
  hbs.registerHelper('t', (cle: string, options: HelperOptions) => traduire(langueDe(options), cle, options.hash));

   // {{dateLongue when}}: "Wednesday, September 23, 2026" / "Wednesday, September 23, 2026"
  hbs.registerHelper('dateLongue', (quand: string, options: HelperOptions) =>
    new Intl.DateTimeFormat(LOCALES[langueDe(options)], { dateStyle: 'full' }).format(new Date(quand)),
  );

   // {{currency price}}: "12.50 €" / "€12.50"
  hbs.registerHelper('monnaie', (montant: number, options: HelperOptions) =>
    new Intl.NumberFormat(LOCALES[langueDe(options)], { style: 'currency', currency: 'EUR' }).format(montant),
  );

  hbs.registerHelper('eq', (a: unknown, b: unknown) => a === b);
}
    • lines 9–12: The page’s language is in the root template ([options.data.root.lang]), where the [main.ts] middleware placed it via [res.locals];
    • line 16: the helper [t] translates a key. The named parameters of the helper ([options.hash]) are the text parameters: [{{t 'ACCUEIL.BONJOUR' nom=nom}}];
    • lines 19–21 and 24–26: dates and amounts are formatted according to the language.

    import { NestFactory } from '@nestjs/core';
import { NestExpressApplication } from '@nestjs/platform-express';
import * as cookieParser from 'cookie-parser';
import { NextFunction, Request, Response } from 'express';
import * as hbs from 'hbs';
import { dirname, join } from 'path';
import { AppModule } from './app.module';
import { enregistrerHelpers } from './helpers';
import { estUneLangue, LANGUE_PAR_DEFAUT } from './i18n';

const RACINE = join(__dirname, '..');

async function bootstrap() {
  const app = await NestFactory.create<NestExpressApplication>(AppModule);
  app.useStaticAssets(join(dirname(require.resolve('bootstrap/package.json')), 'dist'), { prefix: '/bootstrap/' });
  app.setBaseViewsDir(join(RACINE, 'views'));
  app.setViewEngine('hbs');
  enregistrerHelpers(hbs);
  await new Promise<void>((fini) => hbs.registerPartials(join(RACINE, 'views', 'partials'), () => fini()));
  app.use(cookieParser());
   // the language of each page: the language specified in the [langue] cookie, or French otherwise
  app.use((req: Request, res: Response, next: NextFunction) => {
    const cookie = req.cookies?.langue;
    res.locals.lang = estUneLangue(cookie) ? cookie : LANGUE_PAR_DEFAUT;
    next();
  });
  await app.listen(3000);
  console.log('Serveur démarré sur http://localhost:3000');
}

bootstrap();

    Lines 22–26: For each request, the language is read from the [langue] cookie (French by default) and stored in [res.locals.lang], and thus in the template for each view.

    import { Controller, Get, NotFoundException, Param, Req, Res } from '@nestjs/common';
import { Request, Response } from 'express';
import { estUneLangue } from './i18n';

@Controller('langue')
export class LangueController {
   // GET /language/en: saves the selection in a cookie, then returns to the previous page
  @Get(':langue')
  changer(@Param('langue') langue: string, @Req() req: Request, @Res() res: Response): void {
    if (!estUneLangue(langue)) {
      throw new NotFoundException();
    }
    res.cookie('langue', langue, { maxAge: 365 * 24 * 3600 * 1000, sameSite: 'lax', httpOnly: true });
    res.redirect(pageDOrigine(req));
  }
}

// the page you came from (Referer header), of which only the path is retained:
// Redirecting to a full URL from the request would allow
// to send the user to another site (“open redirect” vulnerability)
function pageDOrigine(req: Request): string {
  const referer = req.get('referer');
  if (!referer) {
    return '/';
  }
  try {
    const url = new URL(referer);
    return url.pathname + url.search;
  } catch {
    return '/';
  }
}
    • lines 8–15: [GET /langue/en] stores the selection in a cookie valid for one year, then redirects to the page the user came from. An unknown language results in a 404 error;
    • lines 21–32: The original page is specified by the [Referer] header, which the browser adds when a link is clicked. We retain only the path: redirecting to a full URL from the request would allow a malicious site to create a link that, after passing through our site, sends the user to the malicious site (a vulnerability known as [open redirect]).

@Controller()

     // titrePage is a CLÉ: the view translates it

  @Get('produits')

}
