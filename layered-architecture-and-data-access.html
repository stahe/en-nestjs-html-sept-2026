    9. Layered Architecture and Data Access

    The previous examples kept their data in memory. A real application stores it in a database and is organized into layers (see the diagram in the introduction):

    • the web layer (controllers, views): read the request, call the business layer, select the view;
    • the business layer: business rules;
    • the DAO layer: database queries.

    Each layer is aware only of the layer below it, and as few details about it as possible: the web layer is unaware of the SQL layer, and the business layer is unaware of the HTML layer. This separation makes each layer easier to understand, test, and evolve. Example 28 applies this to managing a list of people: it is, on a smaller scale, the management of doctors and clients from the case study.

    import { Injectable } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { EntityManager, Not, Repository } from 'typeorm';
import { Personne } from '../entities/personne.entity';

// the layer of ACCÈS AUX DONNÉES: queries, no business rules
@Injectable()
export class PersonnesDao {
  constructor(@InjectRepository(Personne) private readonly repository: Repository<Personne>) {}

   // In a transaction, you must go through the transaction “manager”
  private repo(manager?: EntityManager): Repository<Personne> {
    return manager ? manager.getRepository(Personne) : this.repository;
  }

  lister(): Promise<Personne[]> {
    return this.repository.find({ order: { nom: 'ASC', prenom: 'ASC' } });
  }

  lire(id: number, manager?: EntityManager): Promise<Personne | null> {
    return this.repo(manager).findOneBy({ id });
  }

   // a person (with an ID different from [saufId]) with this first and last name
  homonyme(prenom: string, nom: string, saufId: number, manager?: EntityManager): Promise<Personne | null> {
    return this.repo(manager).findOneBy({ prenom, nom, id: Not(saufId) });
  }

   // INSERT if the person has no ID; otherwise, UPDATE
  sauvegarder(personne: Personne, manager?: EntityManager): Promise<Personne> {
    return this.repo(manager).save(personne);
  }

  async supprimer(id: number, manager?: EntityManager): Promise<void> {
    await this.repo(manager).delete(id);
  }
}
    • line 9: [TypeORM] provides a repository ([Repository<Personne>]) for each entity that can read from and write to the table. [@InjectRepository] injects it;
    • lines 12–14: Within a transaction, all queries must go through the same [EntityManager] object—the one associated with the transaction. Each method therefore accepts an optional [manager];
    • lines 16–18: [find] with a sort: [SELECT ... ORDER BY nom, prenom];
    • lines 25–27: a person with the same first and last name, other than [saufId] ([Not]): useful for modifying a record, which must not conflict with itself;
    • lines 30–32: [save] becomes [INSERT] if the object has no identifier, and [UPDATE] otherwise.

    The DAO layer contains no rules: only queries.

    import { Injectable } from '@nestjs/common';
import { DataSource, EntityManager } from 'typeorm';
import { PersonnesDao } from '../dao/personnes.dao';
import { Personne } from '../entities/personne.entity';
import { MetierException } from './metier.exception';

export interface DonneesPersonne {
  prenom: string;
  nom: string;
  age: number;
}

// the MÉTIER layer: the business rules, and only those rules
@Injectable()
export class PersonnesMetier {
  constructor(
    private readonly dao: PersonnesDao,
    private readonly dataSource: DataSource,
  ) {}

  lister(): Promise<Personne[]> {
    return this.dao.lister();
  }

  async lire(id: number): Promise<Personne> {
    const personne = await this.dao.lire(id);
    if (!personne) {
      throw new MetierException("Cette personne n'existe pas (ou plus).");
    }
    return personne;
  }

  ajouter(donnees: DonneesPersonne): Promise<Personne> {
    return this.transaction(async (manager) => {
       // Rule 1: No two people can have the same first and last names
      if (await this.dao.homonyme(donnees.prenom, donnees.nom, 0, manager)) {
        throw new MetierException('Une personne porte déjà ces prénom et nom.');
      }
      const personne = new Personne();
      personne.prenom = donnees.prenom;
      personne.nom = donnees.nom;
      personne.age = donnees.age;
      return this.dao.sauvegarder(personne, manager);
    });
  }

  modifier(id: number, version: number, donnees: DonneesPersonne): Promise<Personne> {
    return this.transaction(async (manager) => {
      const personne = await this.dao.lire(id, manager);
      if (!personne) {
        throw new MetierException("Cette personne n'existe pas (ou plus).");
      }
       // Rule 2: Do not modify a record that has been modified in the meantime by someone else
      if (personne.version !== version) {
        throw new MetierException('Cette fiche a été modifiée entre-temps : rechargez-la.');
      }
      if (await this.dao.homonyme(donnees.prenom, donnees.nom, id, manager)) {
        throw new MetierException('Une personne porte déjà ces prénom et nom.');
      }
      personne.prenom = donnees.prenom;
      personne.nom = donnees.nom;
      personne.age = donnees.age;
      return this.dao.sauvegarder(personne, manager);
    });
  }

  async supprimer(id: number): Promise<Personne> {
    const personne = await this.lire(id);
    await this.dao.supprimer(id);
    return personne;
  }

   // [travail] runs within a transaction: COMMIT if it succeeds, ROLLBACK
   // if it throws an exception. A violation of the uniqueness constraint (two
   // simultaneous insertions, for example) results in a MetierException
  private async transaction<R>(travail: (manager: EntityManager) => Promise<R>): Promise<R> {
    try {
      return await this.dataSource.transaction(travail);
    } catch (e) {
      const erreurSql = e as { code?: string; message?: string };
      if (erreurSql.code === 'ER_DUP_ENTRY' && erreurSql.message?.includes('unq_personnes_prenom_nom')) {
        throw new MetierException('Une personne porte déjà ces prénom et nom.');
      }
      throw e;
    }
  }
}

    Let’s break down this code:

    • lines 16–19: The business layer uses DAO, and [DataSource] from [TypeORM] (the database connection) to open transactions;
    • lines 25–31: an absent person is a [MetierException];
    • lines 33–45: adding a person within a transaction. Rule: no two people may have the same first and last names. This is verified before the write operation to display a clear message;
    • lines 47–65: modification. Additional rule: the optimistic lock (lines 54–56): the modification form carries the version of the record as it was when the form was opened. If it has changed in the meantime (another user has modified the same record), the operation is rejected rather than silently overwriting the user’s work;
    • lines 76–86: [dataSource.transaction(travail)] opens a transaction, executes [travail], then validates it ([COMMIT]) or rolls it back ([ROLLBACK]) if [travail] throws an exception. Lines 81–82: Preliminary checking for homonyms is not always sufficient: two users may add the same person at the same time, and both may pass the check. The database’s uniqueness constraint, however, does not allow any exceptions; the error MySQL [ER_DUP_ENTRY], identified by the constraint, then becomes the same [MetierException].

    import { Transform, Type } from 'class-transformer';
import { IsInt, IsNotEmpty, Max, MaxLength, Min } from 'class-validator';

const nettoyer = ({ value }: { value: unknown }) => (typeof value === 'string' ? value.trim() : value);

// the form fields and their constraints (lengths = SQL columns)
export class FormulairePersonneDto {

    import { Body, Controller, Get, Param, ParseIntPipe, Post, Redirect, Render, Res } from '@nestjs/common';
import { Response } from 'express';
import { MetierException } from '../metier/metier.exception';
import { PersonnesMetier } from '../metier/personnes.metier';
import { flash } from './flash';
import { FormulairePersonneDto } from './formulaire-personne.dto';
import { validerFormulaire } from './validation';

// the WEB layer: read the request, call the business layer, select the view
@Controller()
export class PersonnesController {
  constructor(private readonly metier: PersonnesMetier) {}

  @Get()
  @Redirect('/personnes')
  racine(): void {}

  @Get('personnes')
  @Render('liste')
  async liste(): Promise<object> {
    return { personnes: await this.metier.lister() };
  }

  @Get('personnes/ajouter')
  @Render('formulaire')
  formulaireAjout(): object {
    return { titre: 'Ajouter une personne', action: '/personnes/ajouter', valeurs: {}, erreurs: {} };
  }

  @Post('personnes/ajouter')
  async ajouter(@Body() corps: Record<string, string>, @Res() res: Response): Promise<void> {
    const vue = { titre: 'Ajouter une personne', action: '/personnes/ajouter' };
    const { dto, erreurs } = await validerFormulaire(FormulairePersonneDto, corps);
    if (Object.keys(erreurs).length > 0) {
      res.render('formulaire', { ...vue, valeurs: corps, erreurs, erreur: 'Le formulaire contient des erreurs.' });
      return;
    }
    try {
      const p = await this.metier.ajouter(dto);
      flash(res, { type: 'success', texte: `${p.prenom} ${p.nom} a été ajouté(e).` });
      res.redirect('/personnes');
    } catch (e) {
       // A business rule rejected the addition: the form is redisplayed
      if (!(e instanceof MetierException)) throw e;
      res.render('formulaire', { ...vue, valeurs: corps, erreurs: {}, erreur: e.message });
    }
  }

  @Get('personnes/:id/modifier')
  @Render('formulaire')
  async formulaireModification(@Param('id', ParseIntPipe) id: number): Promise<object> {
    const p = await this.metier.lire(id);
    return { titre: 'Modifier une personne', action: `/personnes/${id}/modifier`, valeurs: p, erreurs: {} };
  }

  @Post('personnes/:id/modifier')
  async modifier(
    @Param('id', ParseIntPipe) id: number,
    @Body() corps: Record<string, string>,
    @Res() res: Response,
  ): Promise<void> {
    const vue = { titre: 'Modifier une personne', action: `/personnes/${id}/modifier` };
    const { dto, erreurs } = await validerFormulaire(FormulairePersonneDto, corps);
    if (Object.keys(erreurs).length > 0) {
      res.render('formulaire', { ...vue, valeurs: corps, erreurs, erreur: 'Le formulaire contient des erreurs.' });
      return;
    }
    try {
      const p = await this.metier.modifier(id, Number(corps.version), dto);
      flash(res, { type: 'success', texte: `${p.prenom} ${p.nom} a été modifié(e).` });
      res.redirect('/personnes');
    } catch (e) {
      if (!(e instanceof MetierException)) throw e;
      res.render('formulaire', { ...vue, valeurs: corps, erreurs: {}, erreur: e.message });
    }
  }

  @Post('personnes/:id/supprimer')
  async supprimer(@Param('id', ParseIntPipe) id: number, @Res() res: Response): Promise<void> {
    try {
      const p = await this.metier.supprimer(id);
      flash(res, { type: 'success', texte: `${p.prenom} ${p.nom} a été supprimé(e).` });
    } catch (e) {
      if (!(e instanceof MetierException)) throw e;
       // No form to redisplay: the error is also displayed instantly
      flash(res, { type: 'danger', texte: e.message });
    }
    res.redirect('/personnes');
  }
}
